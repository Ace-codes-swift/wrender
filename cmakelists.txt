cmake_minimum_required(VERSION 3.20)
project(WRender LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

include(FetchContent)

FetchContent_Declare(
    SDL3
    GIT_REPOSITORY https://github.com/libsdl-org/SDL.git
    GIT_TAG main
)




FetchContent_MakeAvailable(SDL3)

if(CMAKE_PROJECT_NAME STREQUAL PROJECT_NAME)
    # Standalone → executable
    add_executable(WRender
        src/WRender.cpp
    )

    target_link_libraries(WRender PUBLIC SDL3::SDL3)
    target_include_directories(WRender PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)

else()
    # Used as dependency → library
    add_library(WRenderLib
        src/WRender.cpp
    )

    target_link_libraries(WRenderLib PUBLIC SDL3::SDL3)
    target_include_directories(WRenderLib PUBLIC include)

    target_compile_definitions(WRenderLib PRIVATE WRender_IsDependency)
     install(TARGETS WRenderLib
        ARCHIVE DESTINATION lib
        LIBRARY DESTINATION lib
        RUNTIME DESTINATION bin
    )

    install(DIRECTORY include/ DESTINATION include)
endif()
