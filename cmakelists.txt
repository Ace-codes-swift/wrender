cmake_minimum_required(VERSION 3.20)
project(WRender LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)


include(FetchContent)

set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON  CACHE BOOL "" FORCE)

FetchContent_Declare(
    SDL3
    GIT_REPOSITORY https://github.com/libsdl-org/SDL.git
    GIT_TAG release-3.4.4
)

FetchContent_Declare(
    JSON
    GIT_REPOSITORY https://github.com/nlohmann/json.git
    GIT_TAG v3.11.3
)

FetchContent_Declare(
    GLM
    GIT_REPOSITORY https://github.com/g-truc/glm.git
    GIT_TAG 1.0.1
)



FetchContent_MakeAvailable(SDL3)
FetchContent_MakeAvailable(JSON)
FetchContent_MakeAvailable(GLM)

# Always build the core library (no SDL callbacks, no SDL_main)
add_library(WRenderLib
    src/WRenderCore.cpp
    src/3dEngine.cpp
)

target_link_libraries(WRenderLib PUBLIC SDL3::SDL3-static nlohmann_json::nlohmann_json glm::glm)
target_include_directories(WRenderLib PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)

if(CMAKE_PROJECT_NAME STREQUAL PROJECT_NAME)
    # Standalone → executable (SDL callbacks that call into WRenderLib)
    add_executable(WRender
        src/WRender.cpp
        
    )

    target_link_libraries(WRender PRIVATE WRenderLib SDL3::SDL3-static)
    target_include_directories(WRender PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)

else()
    # Used as dependency → install the library
    install(TARGETS WRenderLib
        ARCHIVE DESTINATION lib
        LIBRARY DESTINATION lib
        RUNTIME DESTINATION bin
    )

    install(DIRECTORY include/ DESTINATION include)
endif()
