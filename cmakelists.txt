cmake_minimum_required(VERSION 3.20)
project(WRender LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

include(FetchContent)

FetchContent_Declare(
    SDL3
    GIT_REPOSITORY https://github.com/libsdl-org/SDL.git
    GIT_TAG main
)




FetchContent_MakeAvailable(SDL3)

add_executable(WRender
    src/WRender.cpp
)

target_include_directories(WRender PUBLIC include)

target_link_libraries(WRender PUBLIC SDL3::SDL3)


if(NOT CMAKE_PROJECT_NAME STREQUAL PROJECT_NAME)
    target_compile_definitions(WRender PRIVATE WRender_IsDependency)


endif()
