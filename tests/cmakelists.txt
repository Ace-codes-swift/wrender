cmake_minimum_required(VERSION 3.20)
project(WRender LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

include(FetchContent)

FetchContent_Declare(
    MyHelper
    GIT_REPOSITORY https://github.com/yourusername/your-helper-repo.git
    GIT_TAG        main  # or a specific commit/tag for stability
)

FetchContent_MakeAvailable(MyHelper)

if(NOT CMAKE_PROJECT_NAME STREQUAL PROJECT_NAME)
    target_compile_definitions(MyHelper PRIVATE HELPER_NO_WINDOW)
endif()
